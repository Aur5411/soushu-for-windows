// ==UserScript==
// @name         示例 · 强制深色模式
// @version      1.0.0
// @description  演示 txt 格式导入：这个文件后缀是 .txt，导入后会被自动转成 .user.js
// @match        *://*/*
// @run-at       document-start
// ==/UserScript==

(function () {
  'use strict';

  GM_addStyle(
    'html{filter:invert(1) hue-rotate(180deg);background:#111 !important;}' +
      'img,video,iframe,[style*="background-image"]{filter:invert(1) hue-rotate(180deg);}'
  );

  GM_log('深色模式已应用');
})();
